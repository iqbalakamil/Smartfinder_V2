                document.addEventListener("DOMContentLoaded", function() {
                    const cctvSelect = document.getElementById("cctvSelect");
                    const cctvForm = document.getElementById("cctvForm");
                    const videoContainer = document.getElementById("video");
                    let player;

                    // Fetch daftar CCTV dari server secara langsung
                    fetch("https://eofficev2.bekasikota.go.id/backupcctv/m3/list.json").then(response => response.json()).then(data => {
                        let no = 1;
                        data.forEach(item => {
                            let option = document.createElement("option");
                            option.value = item.path;
                            option.textContent = `${no}. ${item.name.replace(/_/g, ' ')}`;
                            cctvSelect.appendChild(option);
                            no++;
                        }
                        );
                    }
                    ).catch(error => console.error("Gagal memuat data CCTV:", error));

                    // Handle submit form
                    cctvForm.addEventListener("submit", function(event) {
                        event.preventDefault();
                        const selectedCCTV = cctvSelect.value;

                        if (!selectedCCTV)
                            return;

                        const url = `https://eofficev2.bekasikota.go.id/backupcctv/m3/${encodeURIComponent(selectedCCTV)}`;

                        // Hapus player lama jika ada
                        if (player) {
                            player.destroy();
                            videoContainer.innerHTML = "";
                        }

                        // Buat player baru
                        player = new CBPlayer({
                            container: videoContainer,
                            autoplay: true,
                            hotkey: true,
                            video: {
                                url
                            }
                        });

                        player.on("fullscreen", function() {
                            if (/Android|webOS|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent)) {
                                screen.orientation.lock("landscape");
                            }
                        });
                    });
                });

            <script>
                document.addEventListener('DOMContentLoaded', function() {
                    const slider = document.getElementById('programSlider');
                    const prevBtn = document.getElementById('prevBtn');
                    const nextBtn = document.getElementById('nextBtn');
                    let currentIndex = 0;
                    let autoSlideInterval;

                    // Fungsi update posisi slider
                    function updateSlider() {
                        const cardWidth = slider.children[0].getBoundingClientRect().width + 20;
                        // 20 = gap antar card
                        const visibleCards = getVisibleCards();
                        // responsif
                        const maxIndex = Math.ceil(slider.children.length / visibleCards) - 1;

                        // Loop agar auto slider balik ke awal
                        if (currentIndex > maxIndex)
                            currentIndex = 0;

                        const offset = -(cardWidth * visibleCards * currentIndex);
                        slider.style.transform = `translateX(${offset}px)`;
                    }

                    // Fungsi untuk menentukan berapa card tampil (responsif)
                    function getVisibleCards() {
                        if (window.innerWidth < 640)
                            return 1;
                        // mobile
                        if (window.innerWidth < 1024)
                            return 2;
                        // tablet
                        return 3;
                        // desktop
                    }

                    // Tombol Next dan Prev
                    nextBtn.addEventListener('click', () => {
                        const visibleCards = getVisibleCards();
                        const maxIndex = Math.ceil(slider.children.length / visibleCards) - 1;
                        if (currentIndex < maxIndex) {
                            currentIndex++;
                        } else {
                            currentIndex = 0;
                        }
                        updateSlider();
                        resetAutoSlide();
                    }
                    );

                    prevBtn.addEventListener('click', () => {
                        const visibleCards = getVisibleCards();
                        const maxIndex = Math.ceil(slider.children.length / visibleCards) - 1;
                        if (currentIndex > 0) {
                            currentIndex--;
                        } else {
                            currentIndex = maxIndex;
                        }
                        updateSlider();
                        resetAutoSlide();
                    }
                    );

                    // Auto Slide setiap 4 detik
                    function startAutoSlide() {
                        autoSlideInterval = setInterval( () => {
                            currentIndex++;
                            updateSlider();
                        }
                        , 4000);
                    }

                    function resetAutoSlide() {
                        clearInterval(autoSlideInterval);
                        startAutoSlide();
                    }

                    window.addEventListener('resize', updateSlider);

                    // Inisialisasi
                    updateSlider();
                    startAutoSlide();
                });
            </script>

                document.addEventListener("DOMContentLoaded", () => {
                    const counters = document.querySelectorAll(".counter-number");
                    counters.forEach(counter => {
                        const target = +counter.getAttribute("data-target");
                        let count = 0;
                        const increment = target / 100;

                        const updateCount = () => {
                            count += increment;
                            if (count < target) {
                                counter.innerText = Math.floor(count);
                                requestAnimationFrame(updateCount);
                            } else {
                                counter.innerText = target;
                            }
                        }
                        ;

                        updateCount();
                    }
                    );
                }
                );
